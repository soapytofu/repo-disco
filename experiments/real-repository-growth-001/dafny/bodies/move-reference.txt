  assert swapped[..k-1] == xs[..k-1];
  assert swapped[k..] == [xs[k-1]] + xs[k+1..];
  assert xs[..k] == xs[..k-1] + [xs[k-1]];
  var ys := MoveToFront(swapped, k-1);
  assert ys == [xs[k]] + xs[..k] + xs[k+1..];

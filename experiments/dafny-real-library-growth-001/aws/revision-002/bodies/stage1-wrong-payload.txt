    if |input| < 2 then
      Failure("Missing length")
    else
      var length := U.SeqToUInt16(input[..2]);
      var n := length as nat;
      if |input| < 2 + n then
        Failure("Truncated payload")
      else
        var payload := input[2..2+n];
        var rest := input[2+n..];
        assert |payload| == n;
        assert U.UInt16ToSeq(length) == input[..2];
        assert input == input[..2] + payload + rest;
        Success(Frame([], rest))
